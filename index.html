document.addEventListener("DOMContentLoaded", function () { const song = document.getElementById("song"); const playBtn = document.getElementById("playBtn"); const musicStatus = document.getElementById("musicStatus"); const nameInput = document.getElementById("nameInput"); const generateBtn = document.getElementById("generateBtn"); const result = document.getElementById("result"); const generatedLink = document.getElementById("generatedLink"); const copyBtn = document.getElementById("copyBtn"); const whatsappBtn = document.getElementById("whatsappBtn"); const shareBtn = document.getElementById("shareBtn"); /* ================================= MUSIC ================================= */ function playMusic() { song.play() .then(function () { playBtn.textContent = "⏸ Pause Song"; musicStatus.textContent = "🎵 Navratri Special चल रहा है"; }) .catch(function () { musicStatus.textContent = "🔊 Song चलाने के लिए Play दबाएँ"; }); } function pauseMusic() { song.pause(); playBtn.textContent = "▶ Play Song"; musicStatus.textContent = "⏸ Song paused"; } playBtn.addEventListener("click", function () { if (song.paused) { playMusic(); } else { pauseMusic(); } }); song.addEventListener("ended", function () { playBtn.textContent = "▶ Play Song"; musicStatus.textContent = "🎵 Song समाप्त हो गया"; }); /* Browser autoplay policy: Page खुलते ही music play करने की कोशिश। Mobile Chrome इसे रोक सकता है। */ setTimeout(function () { song.play() .then(function () { playBtn.textContent = "⏸ Pause Song"; musicStatus.textContent = "🎵 Navratri Special चल रहा है"; }) .catch(function () { console.log( "Autoplay blocked by browser." ); }); }, 500); /* ================================= READ NAME FROM URL ================================= */ const params = new URLSearchParams( window.location.search ); const senderName = params.get("name"); if (senderName) { const decodedName = decodeURIComponent(senderName); const heading = document.querySelector(".header h1"); if (heading) { heading.innerHTML = "🎁 " + escapeHTML(decodedName) + " ने आपके लिए नवरात्रि का Gift भेजा है 🎁"; } document.title = decodedName + " ने आपके लिए नवरात्रि का Gift भेजा है 🎁"; } /* ================================= GENERATE GIFT LINK ================================= */ generateBtn.addEventListener( "click", function () { const name = nameInput.value.trim(); if (!name) { alert( "कृपया पहले अपना नाम लिखें।" ); nameInput.focus(); return; } /* Current page का URL लिया जा रहा है। Example: https://example.github.io/Special-Gift-/ बनेगा: https://example.github.io/Special-Gift-/?name=Prem%20Singh */ const baseURL = window.location.origin + window.location.pathname; const giftLink = baseURL + "?name=" + encodeURIComponent(name); generatedLink.value = giftLink; result.style.display = "block"; result.scrollIntoView({ behavior: "smooth", block: "center" }); } ); /* ================================= COPY LINK ================================= */ copyBtn.addEventListener( "click", async function () { const link = generatedLink.value; try { await navigator.clipboard.writeText( link ); copyBtn.textContent = "✅ Copied!"; setTimeout(function () { copyBtn.textContent = "📋 Copy Link"; }, 2000); } catch (error) { generatedLink.select(); document.execCommand( "copy" ); copyBtn.textContent = "✅ Copied!"; } } ); /* ================================= WHATSAPP ================================= */ whatsappBtn.addEventListener( "click", function () { const link = generatedLink.value; const message = "🎁 मैंने आपके लिए नवरात्रि का Special Gift भेजा है 🙏🌺\n\n" + "यहाँ क्लिक करके Gift खोलें 👇\n\n" + link + "\n\n" + "🌺 जय माता दी 🌺"; const whatsappURL = "https://wa.me/?text=" + encodeURIComponent(message); window.open( whatsappURL, "_blank" ); } ); /* ================================= NATIVE SHARE ================================= */ shareBtn.addEventListener( "click", async function () { const link = generatedLink.value; const shareData = { title: "नवरात्रि का Special Gift 🎁", text: "🙏 आपके लिए नवरात्रि का Special Gift है 🌺", url: link }; if ( navigator.share ) { try { await navigator.share( shareData ); } catch (error) { console.log( "Share cancelled." ); } } else { try { await navigator.clipboard.writeText( link ); alert( "Gift Link copy हो गया है। अब आप इसे WhatsApp, Instagram, Telegram आदि पर भेज सकते हैं।" ); } catch (error) { alert( link ); } } } ); /* ================================= HTML ESCAPE ================================= */ function escapeHTML(text) { const div = document.createElement("div"); div.textContent = text; return div.innerHTML; } });
